### `t99-spike.mts`

```ts
/**
 * T99 pre-flight spike: does Vitrinee's facilitator settle an x402 `exact`
 * payment whose payer is a `policy_rail` (C…) and whose payTo is Vitrinee's
 * merchant account?
 *
 * Buyer side is AgentPey's real code (PolicyRailStellarScheme + x402Client,
 * exactly what executeBazaarPayment uses). Seller side is Vitrinee's real code
 * (createFacilitatorClient + createX402Server): the requirements are built by
 * Vitrinee's own resource server, then verified and settled through it.
 *
 * Secrets are read from the two env files and never printed.
 */
import { readFileSync } from "node:fs";

const ROOT = "/Users/vicentewolde/dev/AgentPay";

function readEnv(path: string): Record<string, string> {
  const out: Record<string, string> = {};
  for (const line of readFileSync(path, "utf8").split("\n")) {
    const m = /^([A-Z0-9_]+)=(.*)$/.exec(line.trim());
    if (m === null) continue;
    out[m[1]!] = m[2]!.replace(/^"(.*)"$/, "$1");
  }
  return out;
}

const buyerEnv = readEnv(`${ROOT}/.env.local`);
const sellerEnv = readEnv(`${ROOT}/.env.vitrinee.local`);

const { PolicyRailStellarScheme } = await import(`${ROOT}/apps/agent/src/payment/policy-rail-payer.ts`);
const { readRailUsdcBalance } = await import(`${ROOT}/apps/agent/src/policy/rail-balance.ts`);
const { x402Client, x402HTTPClient } = await import(`${ROOT}/apps/agent/node_modules/@x402/core/dist/esm/client/index.mjs`);
const { loadConfig } = await import(`${ROOT}/packages/vitrinee-gateway/src/config.ts`);
const { createFacilitatorClient, createX402Server } = await import(`${ROOT}/packages/vitrinee-gateway/src/x402.ts`);
const { SettlementLedger } = await import(`${ROOT}/packages/vitrinee-gateway/src/settlements.ts`);
const { payerFromTransactionXdr } = await import(`${ROOT}/packages/vitrinee-gateway/src/payer.ts`);

const NETWORK = "stellar:testnet";
const USDC = "CBIELTK6YBZJU5UP2WWQEUCYKLPU6AUNZ2BQ4WWFEIE3USCIHMXQDAMA";
const AMOUNT_ATOMIC = process.argv[2] ?? "100000"; // 0.01 USDC

const railId = buyerEnv["POLICY_RAIL_CONTRACT_ID"]!;
const ownerSecret = buyerEnv["AGENT_SECRET_KEY"]!;
const config = loadConfig({ ...sellerEnv, ADAPTER: "mock" });

console.log("payer (policy_rail):", railId);
console.log("payTo (Vitrinee merchant):", config.merchant.stellarAccount);
console.log("facilitator:", config.facilitator.url, "apiKey set:", config.facilitator.apiKey !== undefined);
console.log("amount atomic:", AMOUNT_ATOMIC);
console.log("rail USDC before:", await readRailUsdcBalance(railId, USDC));

// Seller: Vitrinee's own resource server builds the requirements.
const facilitator = createFacilitatorClient(config);
const ledger = new SettlementLedger();
const server = createX402Server(facilitator, ledger);
await server.initialize();
const [requirements] = await server.buildPaymentRequirements({
  scheme: "exact",
  network: NETWORK,
  payTo: config.merchant.stellarAccount,
  price: { amount: AMOUNT_ATOMIC, asset: USDC },
  maxTimeoutSeconds: config.checkout.maxTimeoutSeconds,
  extra: { paymentFlow: "upfront" },
});
console.log("requirements:", JSON.stringify(requirements));

// Buyer: AgentPey's policy_rail scheme, wrapped exactly like executeBazaarPayment.
const scheme = new PolicyRailStellarScheme({ contractId: railId, ownerSecret });
const client = x402Client.fromConfig({ schemes: [{ network: NETWORK, client: scheme }] });
const httpClient = new x402HTTPClient(client);
const paymentRequired = {
  x402Version: 2,
  error: "Payment required",
  resource: { url: "http://localhost:4021/checkout/spike", description: "T99 spike", mimeType: "application/json" },
  accepts: [requirements],
};
const payload = await httpClient.createPaymentPayload(paymentRequired);
const xdr = (payload.payload as { transaction: string }).transaction;
console.log("payerFromTransactionXdr (today's Vitrinee):", payerFromTransactionXdr(xdr));

const verified = await server.verifyPayment(payload, requirements);
console.log("verify:", JSON.stringify(verified));
if (!verified.isValid) process.exit(2);

const settled = await server.settlePayment(payload, requirements);
console.log("settle:", JSON.stringify(settled));
console.log("ledger record payer:", ledger.take(xdr)?.payer);
console.log("rail USDC after:", await readRailUsdcBalance(railId, USDC));
```
